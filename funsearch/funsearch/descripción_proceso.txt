def calculate_route_distance(route: tuple[int, ...], distances: ndarray) -> float:
    """
    function to calculate the total distance of a given route.
    sum the distances between cities in the route
    """
    distance = sum(distances[route[i], route[i + 1]] for i in range(len(route) - 1))
    # add the distance from the last city to the first city
    distance += distances[route[-1], route[0]]
    return float(distance)

def run_program(program_path):
    from pathlib import Path
    fname = Path(program_path).name
    temp_path = Path(program_path).parent / f"temp_{fname}"

    with open(program_path, "r", encoding="utf-8") as f:
        content = f.read()

    # Añadir print(evaluate(0)) al final
    with open(temp_path, "w", encoding="utf-8") as f:
        f.write(content.strip() + "\nprint(evaluate(0))\n")

    try:
        result = subprocess.run(
            ["python", str(temp_path)],
            capture_output=True,
            text=True,
            timeout=15
        )
        output_lines = result.stdout.strip().splitlines()
        last_line = output_lines[-1] if output_lines else "NO OUTPUT"
        print("🖨️ Resultado capturado de evaluate(0):", last_line)
    except Exception as e:
        last_line = f"ERROR: {str(e)}"

    # Eliminar archivo temporal
    try:
        temp_path.unlink()
    except:
        pass

    return last_line
